cmake_minimum_required(VERSION 3.24)
# The VERSION file at the repo root is the single source of truth. Edit it, build, and the
# number flows into the About page, the update check and the file properties of the exe.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" HIBERNATION_VERSION LIMIT_COUNT 1)
string(STRIP "${HIBERNATION_VERSION}" HIBERNATION_VERSION)
project(Hibernation VERSION ${HIBERNATION_VERSION} LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/version.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h" @ONLY)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/version.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc" @ONLY)
set(WEBVIEW2_ROOT "$ENV{USERPROFILE}/.nuget/packages/microsoft.web.webview2/1.0.3485.44/build/native"
    CACHE PATH "WebView2 SDK build/native directory")
if(NOT EXISTS "${WEBVIEW2_ROOT}/include/WebView2.h")
  message(FATAL_ERROR "WebView2 SDK not found at ${WEBVIEW2_ROOT}. Set -DWEBVIEW2_ROOT=<path to build/native>.")
endif()
set(WEBVIEW2_INCLUDE "${WEBVIEW2_ROOT}/include")
set(WEBVIEW2_LIB "${WEBVIEW2_ROOT}/x64/WebView2LoaderStatic.lib")
add_executable(Hibernation WIN32 src/main.cpp src/renderer.cpp src/config.cpp src/platform.cpp
    src/media.cpp src/diagnostics.cpp src/lock.cpp src/bridge.cpp src/webview.cpp src/cursor.cpp
    src/update.cpp src/app.rc "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
target_compile_definitions(Hibernation PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(Hibernation PRIVATE /W4 /utf-8 /permissive-)
target_include_directories(Hibernation PRIVATE "${WEBVIEW2_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(Hibernation PRIVATE d3d11 dxgi d3dcompiler user32 gdi32 gdiplus shell32 ole32 oleaut32
    psapi wtsapi32 comdlg32 dwmapi windowsapp bcrypt crypt32 advapi32 shlwapi version winhttp "${WEBVIEW2_LIB}")
set_property(TARGET Hibernation PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
